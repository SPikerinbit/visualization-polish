\begin{algorithm}[H]
\caption{Semantic ORDER BY.}
\label{alg:orderby}
\KwIn{rows $D$, instruction $\phi$, capacity $m$, optional limit $K$}
\KwOut{order $\Pi$, or prefix $\Pi_K$}
\lIf{$K=0$ or $|D|\le 1$}{\Return{the trivial frame}}
\eIf{mode is omitted}{buffer when $K$ is set, else full}{use the explicit mode}
\lIf{ascending and $K$ are both set}{force full}
\textsc{ListwiseRank} each chunk of at most $m$\;
Choose $(k,h)$ by minimum estimated input tokens.
Ties keep LT, then smaller $k$, then smaller $h$.
The expectation itself is computed in \textsc{Finite}, which is not among these files\;
\eIf{mode is buffer}{build the producer tree and drain the root}{materialize each pass}
\textsc{Materialize} always closes the root in a \textbf{finally} block, including when $K$ is reached. That close cancels the pump and closes upstreams\;
\lIf{$K$ is set}{truncate the frame to $K$}
\end{algorithm}

\begin{algorithm}[H]
\caption{One sliding-window round. $I$ is a set of run indexes.}
\label{alg:sw}
\KwIn{buffer, quota $h$ or none, capacity $m$}
\If{one run remains and the others are drained}{emit that buffer in pull order and \Return}
$I\leftarrow$ runs that still occupy the buffer or still have supply\;
\eIf{$h$ is fixed}{quota $h$ with $h|I|\le m$; no borrowing}{quota $\max(1,\lfloor m/|I|\rfloor)$, then borrow the shortfall}
$\mathcal{O}\leftarrow\textsc{ListwiseRank}$ of the real buffer\;
Stop the scan when a run has emitted every row it holds and still has supply\;
Emit that safe prefix, refusing an item once $K$ are pending or yielded\;
The rest of $\mathcal{O}$ stays in the buffer\;
\end{algorithm}

\begin{algorithm}[H]
\caption{Pointer-guided LT. One generator for every $h\ge 1$.}
\label{alg:lt}
\KwIn{leaves, $\phi$, $m$, $h$, optional $K$}
Build an exact binary tree. Do not pad to a power of two.
A leaf stores a run index. An internal node stores a loser segment and a pointer\;
\lIf{$h$ is omitted}{$h\leftarrow\lfloor m/(1+\lceil\log_2 k\rceil)\rfloor$}
Require $(1+\lceil\log_2 k\rceil)\,h\le m$\;
Fill every leaf with at most $h$ real rows\;
\ForEach{block returned by \textsc{InitPlan}}{
  \textsc{ListwiseRank} the real frontier rows and repair run order\;
  Replay each internal node in the block\;
}
Let $S$ be the root segment\;
\While{$S$ is nonempty}{
  Yield rows of $S$ until $K$\;
  \lIf{$K$ is reached}{\Return before the next refresh}
  $S\leftarrow\textsc{RefreshPath}$\;
}
\end{algorithm}

\begin{algorithm}[H]
\caption{Replay one node. Slot 0 and slot 1 are arguments, not run ids.}
\label{alg:replay}
\KwIn{slot 0, slot 1, rank map $\rho$, $h$, two child ids}
Order the real rows by $\rho$.
Pad each slot to $h$ with virtual $+\infty$.
Those tags are not sent to the model\;
Let $c$ be the child whose slot first reaches count $h$\;
Set the pointer to $c$.
The winner is the real rows among the first $h$ positions.
The loser is the other real rows\;
\Return{winner, loser, and pointer together}
\end{algorithm}

\begin{algorithm}[H]
\caption{Refresh. Collect, rank, and write-back stay separate.}
\label{alg:refresh}
Follow the pointer from the root to a leaf and fill that leaf only\;
Rank the new segment plus the stored losers, then repair run order\;
\ForEach{node from that leaf's parent up to the root}{
  Put the incoming segment in the current pointer's slot and the loser in the other\;
  Replace the incoming segment by the replay winner\;
}
\end{algorithm}
