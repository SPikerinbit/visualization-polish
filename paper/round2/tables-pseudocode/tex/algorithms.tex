% Paper abstraction: explicit Full / LIMIT mode and requested ranking direction.

{\footnotesize\raggedright
\noindent\textit{Conventions.}
$\Rank_\phi(X)$ orders real rows best first under $\phi$; zero or one row needs no call.
$\operatorname{take}_q$ reads up to $q$ rows in order ($q=\infty$: drain).
LT forms a rank map $\rho=\mathcal L(X)$: rank $X$ once, then repeatedly take the best-ranked remaining run head in $X$.
L and R denote child slots, not original runs. LIMIT pulls on demand; only the root is truncated.\par}

\algcolstart
\begin{algorithm}[H]
\caption{Plan and execute a semantic external merge.}
\label{alg:semantic_external_merge}
\KwIn{rows $D$, instruction $\phi$, capacity $m$, mode Full or LIMIT, output bound $K$ (default $\infty$)}
\lIf{$D=\varnothing$ or $K=0$}{\Return $[\,]$}
Split $D$ into successive chunks of size $m$ (last may be short); $\mathcal R\leftarrow[\Rank_\phi(C) : C\text{ in chunk order}]$\;
\lIf{$|\mathcal R|=1$}{\Return $\operatorname{take}_K(\mathcal R_1)$}
Among permitted $(f,k,h)$, minimize estimated merge input tokens given run lengths, token profile, mode and $K$\;
\While{$|\mathcal R|>1$}{
  Split $\mathcal R$ into consecutive groups of at most $k$ runs\;
  Form one stream per group, in order: forward a singleton;
  otherwise use SW or LT($h$) below according to $f$\;
  \lIf{Full}{materialize every stream before the next pass}
  $\mathcal R\leftarrow$ the resulting runs or pullable streams\;
}
\Return $\operatorname{take}_K(\mathcal R_1)$\;
\end{algorithm}
\algcolend

\algcolstart
\begin{algorithm}[H]
\caption{SW: refill the window; emit its safe prefix.}
\label{alg:sliding_window_kway_group_merge}
\KwIn{$2\le g\le m$ runs; adaptive quota or fixed $1\le h\le\lfloor m/g\rfloor$}
$B\leftarrow[\,]$\;
\While{some row is buffered or unread}{
  $A\leftarrow$ runs with buffered or unread rows\;
  $q\leftarrow h$ if fixed, else $\lfloor m/|A|\rfloor$\;
  In run order, fill each run’s share of $B$ to $q$ or exhaustion\;
  \lIf{adaptive}{borrow available rows round-robin up to $|A|q$ in $B$}
  \lIf{$B$ is empty}{stop}
  \lIf{only one run remains}{yield its buffered rows in pull order, then its unread tail; stop}
  $O\leftarrow\Rank_\phi(B)$\;
  $t\leftarrow$ first position exhausting the buffered rows of a run
  with unread supply; use $|O|$ if none\;
  Yield $O_{1:t}$ and remove those rows from $B$ without reordering the rest\;
}
\end{algorithm}
\algcolend

\algcolstart
\begin{algorithm}[H]
\caption{LT($h$): block initialization and path refresh.}
\label{alg:loser_tree_kway_group_merge}
\KwIn{$g\ge2$ runs, $h\ge1$, capacity $(1+\lceil\log_2 g\rceil)h\le m$}
Build $T$ by halving the run sequence (left half rounds up); let $F$ be its leaves\;
\ForEach{$u\in F$}{$S_u\leftarrow\operatorname{take}_h(R_u)$}
\While{$F\ne\{\mathrm{root}\}$}{
  Treat $F$ as leaves. Cut into maximal subtrees with at most
  $\lfloor m/h\rfloor$ such leaves each\;
  \ForEach{cut subtree with more than one frontier node}{
    $X\leftarrow$ its frontier segments, concatenated left to right; $\rho\leftarrow\mathcal L(X)$\;
    \ForEach{new internal node $v$ in bottom-up order}{
      $(S_v,\ell_v,p_v)\leftarrow\Replay(v,S_{v_L},S_{v_R},\rho)$\;
    }
  }
  $F\leftarrow$ roots of the cut subtrees\;
}
\While{$S_{\rm root}$ contains real rows}{yield $S_{\rm root}$; $S_{\rm root}\leftarrow\Refresh(T)$}
\end{algorithm}
\algcolend

\algcolstart
\begin{algorithm}[H]
\caption{Replay: divide two slots and assign the pointer.}
\label{alg:replay_segment_node}
\KwIn{node $v$, slots $s_L,s_R$ (at most $h$ rows each), rank map $\rho$}
$Z\leftarrow$ real rows of $s_L,s_R$ sorted by $\rho$, retaining each row's slot\;
Append to $Z$: $h-|s_L|$ virtual L entries, then $h-|s_R|$ virtual R entries\;
$p\leftarrow$ child of the first slot to reach $h$ entries in a scan of $Z$\;
\Return (real rows in $Z_{1:h}$, real rows after $h$, $p$)\;
\end{algorithm}
\algcolend

\algcolstart
\begin{algorithm}[H]
\caption{Refresh: replenish only the pointed path.}
\label{alg:refresh_loser_tree_path}
\KwIn{LT($h$) $T$ after root emission, with stored loser segments and child pointers at internal nodes}
Follow stored pointers from the root of $T$ to leaf $u$, recording the path\;
$S\leftarrow\operatorname{take}_h(R_u)$\;
$X\leftarrow S$ followed by path losers $\ell_v$ in root-to-parent order; $\rho\leftarrow\mathcal L(X)$\;
\ForEach{path node $v$ from parent back to root}{
  Put $S$ in the old pointer's slot and $\ell_v$ in the other slot\;
  $(S,\ell_v,p_v)\leftarrow\Replay(v,s_L,s_R,\rho)$\;
}
\Return $S$\;
\end{algorithm}
\algcolend
