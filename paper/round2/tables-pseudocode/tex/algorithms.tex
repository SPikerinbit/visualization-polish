{\footnotesize\raggedright\hyphenpenalty10000\exhyphenpenalty10000
\noindent\textit{Notation.}\quad
$\phi$ instruction; $m$ listwise capacity; $k$ fan-in; $h$ segment length; $R_i$ run $i$; $\Pi$ output; $K$ optional limit.
\Rank\ uses real rows and is the identity below two rows.
Run-order repair emits the current head \Rank\ placed first, while that run remains in the batch.
$+\infty$ pads a short LT slot and is not emitted.
Replay returns $(w,\ell,p)$; the caller stores $\ell$ and $p$. Slots are not run indexes.
\Expect\ estimates input tokens; the formula is not checked.
Families: SW and LT($h$) only.\par}

\algcolstart
\begin{algorithm}[H]
\caption{Semantic external merge.}
\label{alg:semantic_external_merge}
\KwIn{rows $D$, instruction $\phi$, capacity $m$, optional $K$}
\KwOut{order $\Pi$}
\lIf{$K=0$ or $|D|\le 1$}{\Return the trivial order}
\eIf{$K$ is set and $K<|D|$}{mode $\leftarrow$ LIMIT}{mode $\leftarrow$ Full}
\lIf{the caller set a mode}{use that mode}
\lIf{ascending and $K$ is set}{mode $\leftarrow$ Full}
cut $D$ into chunks of at most $m$\;
make each chunk one run by one \Rank\;
$(f,k,h)\leftarrow\arg\min\Expect(f,k,h)$\;
among SW and LT calls that fit in $m$\;
\lIf{\Expect\ ties}{keep LT, then smaller $k$, then smaller $h$}
\eIf{mode is Full}{
  $\mathcal{R}\leftarrow$ the runs\;
  \While{$|\mathcal{R}|>1$}{
    \For{each contiguous group $G$ of at most $k$ runs}{
      \lIf{$|G|=1$}{forward that run}
      \lElse{materialize $\MergeGroup(G,f,h)$ to completion}
    }
    $\mathcal{R}\leftarrow$ the runs of this pass\;
  }
  $\Pi\leftarrow$ the one remaining run\;
}{
  build those same groups as a tree of pullable runs\;
  \lIf{a group has one run}{forward it}
  \eIf{$K$ is set}{pull until $K$ rows or the root ends}{pull until the root ends}
  close the root, which stops unfinished child pulls\;
  $\Pi\leftarrow$ the pulled rows\;
}
\lIf{ascending}{reverse $\Pi$}
\lIf{$K$ is set}{keep the first $K$ rows of $\Pi$}
\end{algorithm}

\algcolstart
\begin{algorithm}[H]
\caption{SW group merge.}
\label{alg:sliding_window_kway_group_merge}
\KwIn{runs $R_1,\ldots,R_g$, capacity $m$, quota $h$ or none}
\KwOut{a locally ordered stream}
$B\leftarrow\emptyset$\;
\While{$B\neq\emptyset$ or some run still has supply}{
  $I\leftarrow\{i : B$ holds $R_i$, or $R_i$ has supply$\}$\;
  \eIf{$h$ is fixed}{
    from each $i\in I$, pull at most $h-|B\cap R_i|$ rows\;
  }{
    $q\leftarrow\max(1,\lfloor m/|I|\rfloor)$\;
    pull each $i\in I$ until $B$ holds $q$ of its rows\;
    borrow from runs that still have supply until $|B|=|I|q$\;
  }
  \lIf{$B=\emptyset$}{\textbf{break}}
  \If{$B$ holds one run and no other run has supply}{
    emit $B$ in pull order, then the rest of that run\;
    \textbf{break}\;
  }
  \lIf{$|B|\le 1$}{$O\leftarrow B$}
  \lElse{$O\leftarrow\Rank(B)$}
  $t\leftarrow|O|$;\ set each $c_i\leftarrow 0$\;
  \For{$j\leftarrow 1$ \KwTo $|O|$}{
    $i\leftarrow$ run of $O_j$;\ $c_i\leftarrow c_i+1$\;
    \If{$c_i=|B\cap R_i|$ and $R_i$ still has supply}{
      $t\leftarrow j$;\ \textbf{break}\;
    }
  }
  \tcp{safe prefix}
  emit $O_1,\ldots,O_t$;\ $B\leftarrow(O_{t+1},\ldots)$\;
}
\end{algorithm}
\algcolend

\algcolstart
\begin{algorithm}[H]
\caption{LT($h$) group merge.}
\label{alg:loser_tree_kway_group_merge}
\KwIn{$k$ runs, capacity $m$, segment length $h$ or none}
\KwOut{a locally ordered stream}
\If{$k=1$}{emit that run until it ends;\ \Return}
$d\leftarrow\lceil\log_2 k\rceil$\;
\lIf{$h$ is absent}{$h\leftarrow\lfloor m/(1+d)\rfloor$}
require $(1+d)\,h\le m$\;
build an exact binary tree, with no power-of-two padding\;
a leaf is one run;\ an internal node stores a loser and a pointer\;
fill every leaf with at most $h$ real rows\;
$F\leftarrow$ the leaves\;
\While{the root $\notin F$}{
  $C\leftarrow$ highest nodes whose $F$-rows fit in one \Rank\;
  \For{each $v\in C$ that is not yet in $F$}{
    \Rank\ the real rows under $v$;\ let $\rho$ repair run order\;
    $(w,\ell,p)\leftarrow$ Replay each new internal node under $v$, bottom-up, under $\rho$\;
    store $\ell$ and $p$;\ pass $w$ upward\;
  }
  $F\leftarrow C$\;
}
$S\leftarrow$ the root segment\;
\While{$S\neq\emptyset$}{
  emit the rows of $S$\;
  $S\leftarrow\textsc{Refresh}$\;
}
\end{algorithm}
\algcolend

\algcolstart
\begin{algorithm}[H]
\caption{Replay one LT($h$) node.}
\label{alg:replay_segment_node}
\KwIn{slots $s_0,s_1$, children $c_0,c_1$, rank $\rho$, length $h$}
\KwOut{$(w,\ell,p)$}
order the real rows of $s_0\cup s_1$ by $\rho$\;
pad each slot to length $h$ with $+\infty$\;
scan until one slot's count reaches $h$\;
$p\leftarrow$ the child of that slot\;
$w\leftarrow$ the real rows in the first $h$ positions\;
$\ell\leftarrow$ the other real rows\;
\Return{$(w,\ell,p)$}\;
\end{algorithm}
\algcolend

\algcolstart
\begin{algorithm}[H]
\caption{Path refresh for LT($h$).}
\label{alg:refresh_loser_tree_path}
follow the pointer from the root down to a leaf\;
fill only that leaf, with at most $h$ real rows\;
$C\leftarrow$ the new leaf segment, then each loser on the path\;
\Rank$(C)$;\ let $\rho$ repair run order\;
$s\leftarrow$ the new leaf segment\;
\For{each node from its parent through the root}{
  put $s$ in the pointer's slot and the stored loser in the other\;
  $(w,\ell,p)\leftarrow$ Replay under $\rho$\;
  store $\ell$ and $p$;\ $s\leftarrow w$\;
}
\Return{$s$}\;
\end{algorithm}
\algcolend
